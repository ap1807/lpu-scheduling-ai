    34→export function ConflictDetail({ conflict, onBack, onChanged }: { conflict: ConflictListRow; onBack: () => void; onChanged: () => void }) {
    35→  const { data, isLoading, refetch } = useQuery({ queryKey: ['conflict', conflict.id], queryFn: () => fetchConflictDetail(conflict.id), enabled: !conflict.id.startsWith('live-') });
    36→  const [running, setRunning] = useState(false);
    37→  const [note, setNote] = useState('');
    38→  const role = (useSession().data?.user as { role?: string })?.role ?? 'STUDENT';
    39→  const canAnalyze = role === 'ADMIN' || role === 'SCHEDULER';
    40→  const canApprove = role === 'ADMIN';
    41→  const queryClient = useQueryClient();
    42→  const { toast } = useToast();
    43→  const navigate = useAppStore((s) => s.navigate);
    44→
    45→  const agent = useMutation({
    46→    mutationFn: () => runResolutionAgent(conflict.id),
    47→    onSuccess: () => {
    48→      toast({ title: 'Agentic AI analysis complete', description: 'Candidates validated by the engine and awaiting human approval.' });
    49→      refetch();
    50→      queryClient.invalidateQueries({ queryKey: ['overview'] });
    51→      onChanged();
    52→    },
    53→    onError: (e: Error) => toast({ title: 'Analysis failed', description: e.message, variant: 'destructive' }),
    54→  });
    55→
    56→  const decide = async (d: 'APPROVE' | 'REJECT', requestId: string, candidateId?: string) => {
    57→    try {
    58→      const res = await decideResolution(requestId, d, candidateId, note);
    59→      if (d === 'APPROVE' && res.applied && res.verification) {
    60→        toast({
    61→          title: 'Change applied to the official timetable',
    62→          description: `Post-change verification: original conflict ${res.verification.originalConflictResolved ? 'RESOLVED ✓' : 'STILL PRESENT'} · ${res.verification.newConflictsIntroduced.length} new conflict(s) · ${res.verification.postDetection.total} total.`,
    63→        });
    64→      } else if (d === 'REJECT') {
    65→        toast({ title: 'Recommendation rejected', description: 'No changes were applied to the timetable.' });
    66→      } else if (!res.applied) {
    67→        toast({ title: 'Application blocked', description: res.reason ?? 'Engine re-validation failed', variant: 'destructive' });
    68→      }
    69→      refetch();
    70→      queryClient.invalidateQueries({ queryKey: ['conflicts'] });
    71→      queryClient.invalidateQueries({ queryKey: ['overview'] });
    72→      queryClient.invalidateQueries({ queryKey: ['audit'] });
    73→      queryClient.invalidateQueries({ queryKey: ['activities'] });
    74→      onChanged();
    75→    } catch (e) {
    76→      toast({ title: 'Decision failed', description: e instanceof Error ? e.message : 'error', variant: 'destructive' });
    77→    }
    78→  };
    79→
    80→  if (isLoading) {
    81→    return <div className="space-y-3"><Loader2 className="w-6 h-6 animate-spin text-orange-600" /><div className="text-sm text-muted-foreground">Loading conflict analysis…</div></div>;
    82→  }
    83→
    84→  if (conflict.id.startsWith('live-')) {
    85→    return (
    86→      <div className="space-y-4">
    87→        <BackButton onBack={onBack} />
    88→        <Card>
    89→          <CardHeader>
    90→            <CardTitle className="flex flex-wrap items-center gap-2 text-base">
    91→              <TypeBadge type={conflict.conflictType} /> <SeverityBadge severity={conflict.severity} /> <StatusBadge status={conflict.status} />
    92→            </CardTitle>
    93→          </CardHeader>
    94→          <CardContent className="space-y-3 text-sm">
    95→            <div>{conflict.rootCause}</div>
    96→            <div className="text-muted-foreground">{conflict.directImpact}</div>
    97→            <div className="text-xs text-muted-foreground">This conflict was detected live but is not yet persisted — re-run conflict detection to register it.</div>
    98→            {canAnalyze && (
    99→              <Button variant="outline" onClick={async () => {
   100→                const { runDetection } = await import('@/lib/client-api');
   101→                await runDetection();
   102→                queryClient.invalidateQueries({ queryKey: ['conflicts'] });
   103→              }}>Register via engine re-scan</Button>
   104→            )}
   105→          </CardContent>
   106→        </Card>
   107→      </div>
   108→    );
   109→  }
   110→
   111→  const d: CD | undefined = data;
   112→  if (!d) return <EmptyState icon={AlertTriangle} title="Conflict not found" description="It may have been resolved and removed by a re-scan." />;
   113→
   114→  const latestRequest = d.resolutionRequests[0];
   115→  const hasAwaiting = latestRequest?.status === 'AWAITING_APPROVAL';
   116→  const r = conflict.affectedResources;
   117→
   118→  return (
   119→    <div className="space-y-4">
   120→      <BackButton onBack={onBack} />
   121→
   122→      {/* ===== header ===== */}
   123→      <Card className="border-l-4" style={{ borderLeftColor: conflict.severity === 'CRITICAL' ? '#ef4444' : conflict.severity === 'HIGH' ? '#f97316' : '#f59e0b' }}>
   124→        <CardHeader className="pb-2">
   125→          <div className="flex flex-wrap items-center gap-2">
   126→            <span className="font-mono text-xs text-muted-foreground">{d.conflict.code}</span>
   127→            <TypeBadge type={d.conflict.conflictType} />
   128→            <TypeBadge type={d.conflict.category} />
   129→            <SeverityBadge severity={d.conflict.severity} />
   130→            <StatusBadge status={d.conflict.status} />
   131→          </div>
   132→          <CardTitle className="text-base leading-snug mt-2">{d.conflict.rootCause}</CardTitle>
   133→          <CardDescription>{d.conflict.directImpact}</CardDescription>
   134→        </CardHeader>
   135→        <CardContent className="space-y-4">
   136→          {/* involved schedule records */}
   137→          <div>
   138→            <SectionLabel icon={FileSearch} label={`Involved schedule records (${d.activities.length})`} />
   139→            <div className="space-y-1.5">
   140→              {d.activities.map((a) => (
   141→                <div key={a.id} className="glass-panel glass-hover rounded-lg border p-3 flex flex-wrap gap-x-4 gap-y-1 items-center text-sm">
   142→                  <div className="min-w-0 flex-1">
   143→                    <div className="font-semibold">
   144→                      {a.courseCode} {a.sectionCode} <span className="text-muted-foreground font-normal">· {a.courseName}</span>
   145→                    </div>
   146→                    <div className="text-xs text-muted-foreground">
   147→                      {a.day} {a.date} · {a.time} · {a.roomCode}{a.roomName ? ` (${a.roomName})` : ''} · {a.blockCode} · {a.facultyName} · {a.studentGroupCode}
   148→                      {a.roomCapacity !== undefined && ` · ${a.capacityRequired}/${a.roomCapacity} seats`}
   149→                    </div>
   150→                    {a.notes && <div className="text-[11px] text-muted-foreground mt-0.5 italic">📝 {a.notes}</div>}
   151→                  </div>
   152→                  <div className="flex gap-1.5">
   153→                    <TypeBadge type={a.activityType} />
   154→                    <Badge variant="outline" className="font-mono text-[10px]">{a.activityCode}</Badge>
   155→                    <Button
   156→                      variant="ghost"
   157→                      size="sm"
   158→                      className="h-6 text-xs"
   159→                      onClick={() => navigate('schedules', { activityId: a.id })}
   160→                    >
   161→                      View in schedule
   162→                    </Button>
   163→                  </div>
   164→                </div>
   165→              ))}
   166→            </div>
   167→          </div>
   168→
   169→          {/* affected resources */}
   170→          <div>
   171→            <SectionLabel icon={AlertTriangle} label="Affected resources" />
   172→            <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-2 text-xs">
   173→              <ResBox label="Rooms" items={r.rooms} />
   174→              <ResBox label="Faculty" items={r.faculty} />
   175→              <ResBox label="Student groups" items={r.studentGroups} />
   176→              <ResBox label="Courses" items={r.courses} />
   177→              <ResBox label="Academic blocks" items={r.blocks} />
   178→              <ResBox label="Dates" items={d.conflict.affectedResources.dates ?? []} />
   179→            </div>
   180→          </div>
   181→
   182→          {/* cascade chain */}
   183→          <div>
   184→            <SectionLabel icon={GitBranch} label="Cascading impact analysis (engine)" />
   185→            <div className="glass-panel rounded-xl border p-3 space-y-2">
   186→              {d.conflict.cascadingImpact.length === 0 ? (
   187→                <div className="text-xs text-muted-foreground">No cascading effects recorded.</div>
   188→              ) : (
   189→                d.conflict.cascadingImpact.map((n, i) => (
   190→                  <div key={i} className="flex gap-2 items-start">
   191→                    <Badge variant="outline" className={cn('text-[10px] shrink-0 mt-0.5 font-mono', n.level === 1 ? 'border-red-300 bg-red-50 text-red-700 dark:bg-red-950/40' : 'border-slate-300 bg-slate-50 dark:bg-slate-800')}>L{n.level}</Badge>
   192→                    <div className="text-xs min-w-0">
   193→                      <span className="font-semibold">{n.resource}</span> — <span className="text-muted-foreground">{n.description}</span>
   194→                      {n.conflictType && <Badge variant="outline" className="ml-1.5 text-[9px]">{n.conflictType}</Badge>}
   195→                    </div>
   196→                  </div>
   197→                ))
   198→              )}
   199→            </div>
   200→          </div>
   201→        </CardContent>
   202→      </Card>
   203→
   204→      {/* ===== agentic AI resolution workflow ===== */}
   205→      <Card>
   206→        <CardHeader className="pb-3">
   207→          <div className="flex flex-wrap items-center justify-between gap-2">
   208→            <div>
   209→              <CardTitle className="text-base flex items-center gap-2">
   210→                <Bot className="w-5 h-5 text-orange-600" /> Agentic AI Resolution Workflow
   211→              </CardTitle>
   212→              <CardDescription>
   213→                10-stage pipeline: engine facts → AI reasoning → deterministic validation → ranking → human approval
   214→              </CardDescription>
   215→            </div>
   216→            {canAnalyze && (
   217→              <Button
   218→                onClick={() => agent.mutate()}
   219→                disabled={agent.isPending || hasAwaiting}
   220→                className="bg-orange-600 hover:bg-orange-700"
   221→              >
   222→                {agent.isPending ? <Loader2 className="w-4 h-4 mr-2 animate-spin" /> : <Sparkles className="w-4 h-4 mr-2" />}
   223→                {agent.isPending ? 'Running agentic analysis…' : hasAwaiting ? 'Awaiting approval below' : d.resolutionRequests.length > 0 ? 'Re-run AI analysis' : 'Run AI analysis'}
   224→              </Button>
   225→            )}
   226→          </div>
   227→        </CardHeader>
   228→        <CardContent className="space-y-4">
   229→          {d.resolutionRequests.length === 0 && !agent.isPending && (
   230→            <EmptyState
   231→              icon={Bot}
   232→              title="No AI analysis yet"
   233→              description={canAnalyze ? 'Run the agentic workflow: the engine re-verifies the conflict, the AI plans and narrates resolutions, candidates are validated against the complete schedule, ranked, and sent for your approval.' : 'An authorized scheduler or administrator must run the AI analysis.'}
   234→            />
   235→          )}
   236→
   237→          {agent.isPending && <PipelineSkeleton />}
   238→
   239→          {latestRequest && (
   240→            <div className="space-y-4">
   241→              {/* stages */}
   242→              <StagePipeline stages={latestRequest.stages} />
   243→
   244→              {/* narrative */}
   245→              {latestRequest.narrative && (
   246→                <div className="glass-ai rounded-xl border border-orange-200 bg-orange-50/60 dark:bg-orange-950/30 dark:border-orange-900 p-4">
   247→                  <div className="flex items-center gap-2 text-sm font-semibold text-orange-800 dark:text-orange-300 mb-2">
   248→                    <BrainCircuit className="w-4 h-4" /> AI root-cause narrative
   249→                    <Badge variant="outline" className="text-[10px] font-normal">{latestRequest.narrative ? 'engine facts + LLM' : ''}</Badge>
   250→                  </div>
   251→                  <div className="text-sm whitespace-pre-wrap leading-relaxed">{latestRequest.narrative}</div>
   252→                </div>
   253→              )}
   254→
   255→              {/* candidates */}
   256→              <div>
   257→                <SectionLabel icon={ListChecks} label={`Candidate solutions (${latestRequest.candidates.length}) — engine-validated & ranked`} />
   258→                <div className="space-y-2">
   259→                  {latestRequest.candidates.map((c) => (
   260→                    <CandidateCard
   261→                      key={c.id}
   262→                      c={c}
   263→                      isRecommended={c.status === 'SELECTED'}
   264→                      canApprove={canApprove && hasAwaiting}
   265→                      onApprove={() => decide('APPROVE', latestRequest.id, c.id)}
   266→                    />
   267→                  ))}
   268→                </div>
   269→              </div>
   270→
   271→              {/* approval / rejection */}
   272→              {hasAwaiting && (
   273→                <div className="glass-panel rounded-xl border p-4 space-y-3">
   274→                  <div className="flex items-center gap-2 text-sm font-semibold">
   275→                    <UserCheck className="w-4 h-4 text-emerald-600" /> Human approval required
   276→                  </div>
   277→                  <div className="text-xs text-muted-foreground leading-relaxed">
   278→                    The AI cannot modify the official LPU timetable. {canApprove ? 'As System Administrator you may approve the selected recommendation or reject it.' : 'Only a System Administrator can approve; you may reject or request re-analysis.'}
   279→                  </div>
   280→                  <Textarea placeholder="Decision note (recorded in the audit trail)" value={note} onChange={(e) => setNote(e.target.value)} rows={2} />
   281→                  <div className="flex flex-wrap gap-2">
   282→                    {canApprove && (
   283→                      <Button className="bg-emerald-600 hover:bg-emerald-700" onClick={() => decide('APPROVE', latestRequest.id)}>
   284→                        <CheckCircle2 className="w-4 h-4 mr-1.5" /> Approve &amp; apply selected recommendation
   285→                      </Button>
   286→                    )}
   287→                    <Button variant="outline" className="border-red-200 text-red-700 hover:bg-red-50" onClick={() => decide('REJECT', latestRequest.id)}>
   288→                      <XCircle className="w-4 h-4 mr-1.5" /> Reject
   289→                    </Button>
   290→                  </div>
   291→                </div>
   292→              )}
   293→
   294→              {/* decision outcome */}
   295→              {latestRequest.status !== 'AWAITING_APPROVAL' && latestRequest.decidedBy && (
   296→                <div className={cn('glass-tint rounded-xl border p-4 text-sm', latestRequest.status === 'APPLIED' || latestRequest.status === 'APPROVED' ? 'border-emerald-200 bg-emerald-50/60 dark:bg-emerald-950/30 dark:border-emerald-900' : 'border-slate-200 bg-slate-50 dark:bg-slate-900')}>
   297→                  <div className="font-semibold flex items-center gap-2">
   298→                    {latestRequest.status === 'APPLIED' || latestRequest.status === 'APPROVED' ? <CheckCircle2 className="w-4 h-4 text-emerald-600" /> : <XCircle className="w-4 h-4" />}
   299→                    {latestRequest.status === 'APPLIED' ? 'Applied to official schedule' : latestRequest.status === 'REJECTED' ? 'Rejected' : latestRequest.status}
   300→                  </div>
   301→                  <div className="text-xs text-muted-foreground mt-1">
   302→                    Decided by {latestRequest.decidedBy} · {latestRequest.decidedAt ? new Date(latestRequest.decidedAt).toLocaleString() : ''}
   303→                    {latestRequest.decisionNote ? ` — "${latestRequest.decisionNote}"` : ''}
   304→                  </div>
   305→                  {d.conflict.resolutionNote && <div className="text-xs mt-2">{d.conflict.resolutionNote}</div>}
   306→                </div>
   307→              )}
   308→
   309→              {/* history */}
   310→              {d.resolutionRequests.length > 1 && (
   311→                <Accordion type="single" collapsible>
   312→                  <AccordionItem value="history">
   313→                    <AccordionTrigger className="text-sm">
   314→                      <span className="flex items-center gap-2"><History className="w-4 h-4" /> Previous analyses ({d.resolutionRequests.length - 1})</span>
   315→                    </AccordionTrigger>
   316→                    <AccordionContent className="space-y-2">
   317→                      {d.resolutionRequests.slice(1).map((req) => (
   318→                        <div key={req.id} className="rounded border p-3 text-xs space-y-1">
   319→                          <div className="flex items-center gap-2">
   320→                            <span className="font-mono text-muted-foreground">{req.code}</span>
   321→                            <StatusBadge status={req.status} />
   322→                            <span className="text-muted-foreground">{new Date(req.createdAt).toLocaleString()} · by {req.requestedByName}</span>
   323→                          </div>
   324→                          <div className="text-muted-foreground">{req.candidates.length} candidates · decision: {req.decidedBy ?? 'pending'}</div>
   325→                        </div>
   326→                      ))}
   327→                    </AccordionContent>
   328→                  </AccordionItem>
   329→                </Accordion>
   330→              )}
   331→            </div>
   332→          )}
   333→        </CardContent>
   334→      </Card>
   335→
   336→      <div className="text-[11px] text-muted-foreground flex items-center gap-2">
   337→        <Cog className="w-3.5 h-3.5" />
   338→        Engine verification: {d.engineVerification.detectionPass} — {d.engineVerification.verifiedBy}
   339→      </div>
   340→    </div>
   341→  );
   342→}

   411→function CandidateCard({ c, isRecommended, canApprove, onApprove }: {
   412→  c: CD['resolutionRequests'][0]['candidates'][0];
   413→  isRecommended: boolean;
   414→  canApprove: boolean;
   415→  onApprove: () => void;
   416→}) {
   417→  const [showChecks, setShowChecks] = useState(false);
   418→  return (
   419→    <div className={cn(
   420→      'rounded-xl border glass-panel p-3.5 space-y-2.5',
   421→      isRecommended && 'glass-ai border-orange-300 bg-orange-50/50 dark:bg-orange-950/25 dark:border-orange-800 ring-1 ring-orange-200',
   422→      !c.valid && 'opacity-75 border-red-200'
   423→    )}>
   424→      <div className="flex flex-wrap items-center gap-2">
